АРХИТЕКТУРА ПРОГРАММЫ:
Программа состоит из нескольких взаимосвязанных классов:
- Task (Задача) - представляет отдельную задачу
- Column (Колонка) - содержит группу задач  **
- Board (Доска) - управляет колонками и разработчиками
- Developer (Разработчик) - представляет члена команды
- ScrumBoardUI - пользовательский интерфейс
- Json_worker - работа с сохранением/загрузкой
- Manager - функции-помощники для создания сущностей

std::string Task::generate_random_string(int length) {

Реализация метода генерации случайной строки. Принимает длину строки.


const std::string charset = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";

Набор символов для генерации ID. Исключены похожие символы (0/O, 1/I) чтобы избежать путаницы.


std::random_device rd;

Создание объекта random_device - источника энтропии. Использует аппаратный генератор если доступен.


std::mt19937 generator(rd());

Вихрь Мерсенна (Mersenne Twister) - качественный псевдослучайный генератор. Инициализируется значением из random_device.


std::uniform_int_distribution<int> distribution(0, charset.size() - 1);

Равномерное распределение для генерации индексов в charset. Гарантирует что все символы имеют равную вероятность.


std::string result;

Создание пустой строки для результата.


for (int i = 0; i < length; ++i) {
        result += charset[distribution(generator)];
    }

Цикл генерации: distribution(generator) генерирует случайный индекс, charset[...] получает символ, += добавляет его к результату.


return result;

Возврат сгенерированной строки.


std::string Task::generate_id() {

Реализация генерации уникального ID.


std::string new_id;
    bool unique_found = false;

Переменные для хранения нового ID и флага уникальности.


const int max_attempts = 100;
    int attempts = 0;

Ограничение максимального числа попыток - защита от бесконечного цикла.


while (!unique_found && attempts < max_attempts) {

Цикл продолжается пока не найден уникальный ID или не превышено максимальное число попыток.


    new_id = generate_random_string(6);

Генерация ID длиной 6 символов. 6 - компромисс между уникальностью и читаемостью.


    if (std::find(used_ids.begin(), used_ids.end(), new_id) == used_ids.end()) {

Проверка уникальности: std::find ищет new_id в векторе used_ids. Если возвращает used_ids.end() - элемент не найден.


        used_ids.push_back(new_id);
            unique_found = true;

Если ID уникален - добавляем в использованные и устанавливаем флаг.


    attempts++;

Увеличиваем счетчик попыток.


if (!unique_found) {
        throw std::runtime_error("Failed to generate unique task ID");
    }

Если не удалось сгенерировать уникальный ID - бросаем исключение.


return new_id;

Возврат уникального ID.

void Column::add_task(std::unique_ptr<Task> task) {

Реализация добавления задачи.


if (!task) {
        throw std::invalid_argument("Task cannot be null");
    }

Проверка что передан не nullptr. Защита от ошибок.


this->tasks.push_back(std::move(task));

Перемещение задачи в конец вектора. std::move необходим потому что unique_ptr нельзя копировать.


void Column::delete_task(const std::string& task_title) {

Удаление задачи по заголовку.


auto it = std::find_if(this->tasks.begin(), this->tasks.end(),
        [&](const std::unique_ptr<Task>& task) {
            return task->get_title() == task_title;
        });

Поиск задачи с помощью std::find_if и лямбда-функции. Лямбда захватывает task_title по ссылке ([&]) и сравнивает заголовки.


if (it != this->tasks.end()) {
        this->tasks.erase(it);

Если задача найдена (итератор не указывает на конец) - удаляем ее из вектора.


} else {
        throw std::runtime_error("Task not found: " + task_title);
    }

Если не найдена - бросаем исключение с информацией.

Task* Column::find_task(const std::string& title) const {

Поиск задачи в колонке по заголовку.


auto it = std::find_if(tasks.begin(), tasks.end(),
        [&](const std::unique_ptr<Task>& task) {
            return task->get_title() == title;
        });

Аналогично delete_task, используем find_if с лямбдой.


return it != tasks.end() ? it->get() : nullptr;

Тернарный оператор: если нашли - возвращаем указатель (it->get()), иначе nullptr.


void move_task(Column* start, Column* end, Task* task) {

Реализация перемещения задачи между колонками.


if (!start || !end || !task) {
        throw std::invalid_argument("Columns and task cannot be null");
    }

Проверка валидности указателей.


auto& start_tasks = start->get_tasks();

Получение ссылки на задачи исходной колонки.


auto it = std::find_if(start_tasks.begin(), start_tasks.end(),
        [&](const std::unique_ptr<Task>& t) {
            return t.get() == task;
        });

Поиск по указателю (а не по содержимому). Сравниваем именно адреса в памяти.


if (it != start_tasks.end()) {
        auto task_ptr = std::move(*it);

Если нашли - перемещаем задачу во временную переменную.


    start_tasks.erase(it);

Удаляем задачу из исходной колонки.


    end->add_task(std::move(task_ptr));

Добавляем задачу в целевую колонку, передавая владение.


} else {
        throw std::runtime_error("Task not found in source column");
    }

Если не нашли - исключение.


Task* search_task(Board& board, const std::string& col, const std::string& title) {

Поиск задачи на всей доске.


Column* column = board.find_column(col);

Поиск колонки по имени.


if (!column) {
        throw std::runtime_error("Column not found: " + col);
    }

Если колонка не найдена - исключение.


Task* task = column->find_task(title);

Поиск задачи в колонке.


if (!task) {
        throw std::runtime_error("Task not found: " + title + " in column: " + col);
    }

Если задача не найдена - исключение с детальной информацией.


return task;

Возврат найденной задачи.

bool Json_worker::is_valid_board_file(const std::string& file_path) const {

Проверка валидности файла доски.


std::ifstream file(file_path);
    if (!file.is_open()) {
        std::cout << "Cannot open file: " << file_path << std::endl;
        return false;
    }
    
std::stringstream buffer;
    buffer << file.rdbuf();
    file.close();
    
Document temp_doc;
    temp_doc.Parse(buffer.str().c_str());
    
if (temp_doc.HasParseError()) {
        std::cout << "Invalid JSON format" << std::endl;
        return false;
    }
    
if (!temp_doc.IsObject()) {
        std::cout << "Root element is not an object" << std::endl;
        return false;
    }
    
if (temp_doc.MemberBegin() == temp_doc.MemberEnd()) {
        std::cout << "No board found in file" << std::endl;
        return false;
    }

Базовые проверки структуры JSON.


const std::string board_name = temp_doc.MemberBegin()->name.GetString();
    const Value& board_obj = temp_doc[board_name.c_str()];
    
if (!board_obj.IsObject()) {
        std::cout << "Board data is not an object" << std::endl;
        return false;
    }
    
bool has_valid_structure = false;
    
for (Value::ConstMemberIterator itr = board_obj.MemberBegin(); 
         itr != board_obj.MemberEnd(); ++itr) {
        std::string field_name = itr->name.GetString();
        
    if (field_name == "developers" || field_name == "ids") {
            has_valid_structure = true;
            break;
        }
        
    if (itr->value.IsObject()) {
            const Value& tasks_obj = itr->value;
            if (tasks_obj.MemberBegin() != tasks_obj.MemberEnd()) {
                has_valid_structure = true;
                break;
            }
        }
    }
    
return has_valid_structure;
}

        std::filesystem::path path(file_path_input_str);

            for (const auto& entry : std::filesystem::directory_iterator(path)) {
                if (entry.is_regular_file() && entry.path().extension() == ".json") {
                    json_files.push_back(entry.path().filename().string());
                }
            }
        } else if (std::filesystem::is_regular_file(path)) {
            auto parent_path = path.parent_path();
            file_path_input_str = parent_path.string();
            
        for (const auto& entry : std::filesystem::directory_iterator(parent_path)) {
                if (entry.is_regular_file() && entry.path().extension() == ".json") {
                    json_files.push_back(entry.path().filename().string());
                }
            }
        } else {
            auto parent_path = path.parent_path();
            if (std::filesystem::is_directory(parent_path)) {
                file_path_input_str = parent_path.string();
                
            for (const auto& entry : std::filesystem::directory_iterator(parent_path)) {
                    if (entry.is_regular_file() && entry.path().extension() == ".json") {
                        json_files.push_back(entry.path().filename().string());
                    }
                }
            }
        }
    } catch (const std::exception& e) {
        std::cout << "Error reading directory: " << e.what() << std::endl;
    }
    
if (!json_files.empty() && selected_file >= json_files.size()) {
        selected_file = 0;
    } else if (json_files.empty()) {
        selected_file = 0;

Element ScrumBoardUI::render_board() {
    Elements column_elements;
    
    // Проходим по всем колонкам доски
    for (const auto& column : board->get_columns()) {
        Elements task_elements;
        
        // Заголовок колонки с названием
        task_elements.push_back(text(column->get_name()) | bold | center);
        // Разделительная линия под заголовком
        task_elements.push_back(separator());
        
        auto& tasks = column->get_tasks();
        // Обработка пустой колонки
        if (tasks.empty()) {
            // Сообщение о отсутствии задач + занимает пространство
            task_elements.push_back(text("No tasks") | center | flex | size(HEIGHT, EQUAL, 10));
        } else {
            int task_count = tasks.size();
            
            // УРОВНИ ДЕТАЛИЗАЦИИ:
            // Определяем сколько информации показывать в зависимости от количества задач
            int detail_level;
            if (task_count <= 3) detail_level = 4;      // Вся информация
            else if (task_count <= 4) detail_level = 3; // Почти вся информация  
            else if (task_count <= 5) detail_level = 2; // Основная информация
            else if (task_count <= 6) detail_level = 1; // Минимум информации
            else detail_level = 0;                      // Только заголовок
            
            // Высота задачи в зависимости от уровня детализации
            int task_height;
            switch (detail_level) {
                case 4: task_height = 8; break; // Высокая - полная информация
                case 3: task_height = 6; break; // Средняя - описание + разработчик + приоритет
                case 2: task_height = 5; break; // Компактная - разработчик + приоритет
                case 1: task_height = 4; break; // Минимальная - только разработчик
                case 0: task_height = 3; break; // Ультра-компактная - только заголовок
            }
            
            // Отрисовка каждой задачи в колонке
            for (size_t i = 0; i < tasks.size(); ++i) {
                const auto& task = tasks[i];
                
                // Получение имени разработчика
                std::string developer_name = "Unassigned";
                if (task->get_developer()) {
                    developer_name = task->get_developer()->get_name();
                }
                
                Elements task_content;
                
                // ЗАГОЛОВОК ЗАДАЧИ - отображается всегда
                task_content.push_back(text("📝 " + task->get_title()) | bold | center);
                
                // УРОВЕНЬ 1+: Разработчик
                if (detail_level >= 1 && task->get_developer()!=nullptr) {
                    task_content.push_back(separator()); // Разделитель
                    task_content.push_back(text("👨 " + developer_name) | center);
                }
                
                // УРОВЕНЬ 2+: Приоритет
                if (detail_level >= 2 && task->get_priority() != -1) {
                    task_content.push_back(text("🎯 " + std::to_string(task->get_priority())) | center);
                }
                
                // УРОВЕНЬ 3+: Описание (если есть)
                if (detail_level >= 3 && !task->get_description().empty()) {
                    std::string desc = task->get_description();
                    // Обрезаем длинные описания
                    if (desc.length() > 20) desc = desc.substr(0, 17) + "...";
                    task_content.push_back(text("📋 " + desc) | center);
                }
                
                // Создание элемента задачи
                auto task_element = vbox(std::move(task_content)) 
                    | border          // Рамка вокруг задачи
                    | size(HEIGHT, EQUAL, task_height); // Фиксированная высота
                    
                task_elements.push_back(task_element);
                
                // Добавляем отступ между задачами (кроме последней)
                if (i < tasks.size() - 1) task_elements.push_back(filler());
            }
        }
        
        // СОЗДАНИЕ КОЛОНКИ:
        // Каждая колонка получает равную долю пространства
        column_elements.push_back(
            vbox(std::move(task_elements)) 
            | border        // Рамка вокруг колонки
            | flex          // Растягивается по вертикали
            | frame         // Обрамление
            | vscroll_indicator // Индикатор прокрутки если контент не помещается
        );
    }
    
    // ГЛАВНЫЙ КОНТЕЙНЕР:
    // Простой и эффективный способ занять всю ширину
    return vbox({
        // Заголовок доски
        text("SCRUM Board - " + board->get_name()) | bold | hcenter,
        // Разделитель
        separator(),
        // Горизонтальное расположение колонок
        hbox(std::move(column_elements)) 
            | flex    // Растягивается по вертикали
            | xflex   // Растягивается по горизонтали (равномерное распределение колонок)
    }) 
    | flex   // Главный контейнер растягивается
    | xflex; // Занимает всю ширину терминала
}
